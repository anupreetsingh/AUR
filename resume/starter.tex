% For writing bullet points in the resume. Internally makes a nested itemize environment and the arugment is the items of that itemzie environment. with a dash as the bullet point.
\newcommand{\BulletItems}[1]{
  \item
    \begin{itemize}[label={-}, leftmargin=*, labelsep=0.3em, topsep=0pt, partopsep=0pt, itemsep=0pt, parsep=0pt]
      \small #1
    \end{itemize}\vspace{-5pt}
}

% For writing Single Line Subheading items with Four fields (e.g., title, Organization, location, date)
\newcommand{\SLSubheadingItemFF}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} \textbar{} \textit{\small #2} \textbar{} \textit{\small #3} & {\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

% For writing Single Line Subheading items with Three fields (e.g., title, Organization, date)
\newcommand{\SLSubheadingItemThF}[3]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} \textbar{} \textit{\small #2} & {\small #3} \\
    \end{tabular*}\vspace{-7pt}
}

% For writing Single Line Subheading items with Two fields (e.g., title and date)
\newcommand{\SLSubheadingItemTF}[2]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & {\small #2} \\
    \end{tabular*}\vspace{-7pt}
}


% For writing Single Line Subheading items with a Hyperlink (Mainly used for linking to project repositories) with Four fields (e.g., title, link, description, date)
\newcommand{\SLSubheadingItemLink}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{\href{#2}{#1}} \textbar{} \textit{\small #3} & {\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

% For writing Double Line Subheading items with Four fields (e.g., title, date, Organization, location)
\newcommand{\DLSubheadingItem}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & {\small #2} \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}
